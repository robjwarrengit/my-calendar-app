async function loadLiveCalendarEvents(forceRefresh = false) { const params = new URLSearchParams({ stripNotes: state.privacy.stripNotes ? "1" : "0", hideLocation: state.privacy.hideLocation ? "1" : "0", ...(forceRefresh ? { refresh: "1" } : {}) }); const res = await fetch(`/api/calendar/events?${params.toString()}`); if (!res.ok) throw new Error("Failed to load family calendar feed"); return await res.json(); }
